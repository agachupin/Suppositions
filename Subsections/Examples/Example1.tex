\subsection{Squares of Odd Integers}
\label{subsections:examples:example1}



\begin{exercise}{
    Prove that if $a$ is an odd integer, %
    then $a^2 = 8m + 1$ for some $m \in \Z$.
}
\end{exercise}



\begin{proof}{
    This statement can be proven by %
    \hyperref[subsections:techniques:universal:cases]{division into cases}.

    \begin{panel}[Setup.]{
        Let $a$ be an odd integer. %
        Then $\exists k \in \Z$ such that $a = 2k + 1$. %
        Thus, %
        \begin{align*}{
            a^2 &= (2k + 1)^2\\
                &= 4k^2 + 4k + 1\\
                &= 4k(k + 1) + 1.
        }
        \end{align*}
        Either $k$ is even or odd.
    }
    \end{panel}

    \begin{panel}[Case 1: $k$ is even.]{
        Then $\exists n \in \Z$ such that $k = 2n$. %
        Thus, %
        \begin{align*}{
            a^2 &= 4k(k + 1) + 1\\
                &= 4(2n)[(2n) + 1] + 1\\
                &= 4(4n^2 + 2n) + 1\\
                &= 8(2n^2 + n) + 1.
        }
        \end{align*}
        Let $m = 2n^2 + n$. %
        Then $m \in \Z$ and $a^2 = 8m + 1$.
    }
    \end{panel}

    \begin{panel}[Case 2: $k$ is odd.]{
        Then $\exists n \in \Z$ such that $k = 2n + 1$. %
        Thus, %
        \begin{align*}{
            a^2 &= 4k(k + 1) + 1\\
                &= 4(2n + 1)[(2n + 1) + 1] + 1\\
                &= 4(2n + 1)(2n + 2) + 1\\
                &= 4(4n^2 + 6n + 2) + 1\\
                &= 8(2n^2 + 3n + 1) + 1.
        }
        \end{align*}
        Let $m = 2n^2 + 3n + 1$. %
        Then $m \in \Z$ and $a^2 = 8m + 1$.
    }
    \end{panel}

    In either case, $a^2 = 8m + 1$ for some $m \in \Z$. %
    Therefore, the statement holds.
}
\end{proof}

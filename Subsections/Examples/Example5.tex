\subsection{Left Inverses of Injections}
\label{subsections:examples:example5}



\begin{exercise}{
    Let $A$ be a nonempty set. %
    Prove that a function $f \colon A \to B$ is injective if and only if %
    $f$ has a left inverse, that is, %
    a function $g \colon B \to A$ satisfying $\forall a \in A$, %
    $(g \circ f)(a) = a$.
}
\end{exercise}



\begin{proof}{
    This statement can be proven by proving each direction of the %
    biconditional.

    \begin{panel}[Part 1: Proving $f$ has a left inverse.]{
        Let $f \colon A \to B$ be an injective function. %
        Since $A$ is nonempty, $\exists a_0 \in A$. %
        Let $g \colon B \to A$ be the function defined $\forall b \in B$ by %
        \[
            g(b) =
            \begin{cases}
                a & \text{if } \exists a \in A \text{ such that } b = f(a), \\
                a_0 & \text{if } \nexists a \in A \text{ such that } b = f(a).
            \end{cases}
        \]
        Because $f$ is injective, $\forall b \in B$, %
        if $\exists a \in A$ with $b = f(a)$, then $a$ is unique. %
        Thus, the first case gives each such $b$ exactly one output. %
        The second case gives every other $b \in B$ the output $a_0$. %
        Therefore, $g$ is a well-defined function.

        To prove $g$ is a left inverse of $f$, let $a \in A$. %
        By the definition of $g$, %
        \[
            g(f(a)) = a.
        \]
        Because $g(f(a)) = a$ holds $\forall a \in A$, %
        $g$ is a valid left inverse of $f$.
    }
    \end{panel}

    \begin{panel}[Part 2: Proving $f$ is injective.]{
        Let $f \colon A \to B$ be a function and %
        $g \colon B \to A$ a left inverse of $f$. %
        Then $\forall a \in A$, %
        \[
            (g \circ f)(a) = g(f(a)) = a.
        \]
        Let $a_1, a_2 \in A$ with $f(a_1) = f(a_2)$. %
        Then %
        \[
            a_1 = g(f(a_1)) = g(f(a_2)) = a_2.
        \]
        Thus, $\forall a_1, a_2 \in A$, $f(a_1) = f(a_2) \implies a_1 = a_2$. %
        Therefore, $f$ is injective.
    }
    \end{panel}

    Thus, there is a left inverse when $f$ is injective, %
    and $f$ is an injective function when there is a left inverse. %
    Therefore, the statement holds.
}
\end{proof}

\newcommand{\titleuppercase}[1]{\textls[50]{\MakeUppercase{#1}}}

\titleformat{\section}[block]%
    {\large\bfseries\filcenter}%
    {\mdseries\thesection}%
    {1em}%
    {\titleuppercase}

\let\sectionstyle\section
\RenewDocumentCommand{\section}{s o m}{%
    \IfBooleanTF{#1}{%
        \sectionstyle*{#3}%
    }{%
        \clearpage
        \thispagestyle{empty}%
        \vspace*{\fill}%
        \IfNoValueTF{#2}{\sectionstyle{#3}}{\sectionstyle[#2]{#3}}%
        \vspace*{\fill}%
    }%
}

\titleformat{\subsection}[block]%
    {\normalsize\bfseries}%
    {\mdseries\thesubsection}%
    {1em}%
    {\titleuppercase}

\NewTColorBox{panel}{o}{
    enhanced, unbreakable,
    frame hidden, boxrule = 0pt, colback = black!3,
    borderline west = {1pt}{0pt}{black!75},
    sharp corners,
    left = 16pt, right = 16pt, top = 12pt, bottom = 12pt,
    before skip = 14pt, after skip = 14pt,
    before upper = {%
        \IfValueT{#1}{%
            {\small\bfseries\color{black!70}\titleuppercase{#1}\par}%
            \nointerlineskip
            \vspace{7pt}%
            {\color{black!20}\hrule height 0.4pt}%
            \nointerlineskip
            \vspace{7pt}%
        }%
    }
}



\newcommand{\panelheading}[3]{%
    {\small\bfseries\color{#1}\titleuppercase{#3}\par}%
    \nointerlineskip
    \vspace{7pt}%
    {\color{#2}\hrule height 0.4pt}%
    \nointerlineskip
    \vspace{7pt}%
}

\newlength{\annotationwidth}
\setlength{\annotationwidth}{0.36\textwidth}

\tcbset{
    annotated/.style = {
        enhanced, unbreakable,
        frame hidden, boxrule = 0pt, sharp corners,
        sidebyside, sidebyside align = top, lower separated = false,
        sidebyside gap = 32pt, righthand width = \annotationwidth,
        colback = black!3,
        left = 16pt, right = 16pt, top = 12pt, bottom = 12pt,
        before skip = 14pt, after skip = 14pt,
        underlay = {%
            \begin{tcbclipinterior}
                \fill[blue!8]
                    ([xshift = -\annotationwidth - 32pt]interior.north east)
                    rectangle (interior.south east);
            \end{tcbclipinterior}%
        },
        before lower = {%
            \panelheading{blue!50!black}{blue!25}{Annotation.}%
        }
    }
}

\NewDocumentEnvironment{annotatedpanel}{o m m}{%
    \begin{tcolorbox}[
        annotated,
        borderline west = {1pt}{0pt}{black!75},
        before upper = {%
            \IfValueT{#1}{\panelheading{black!70}{black!20}{#1}}%
        }
    ]
        #2
        \tcblower
        \small #3
    \end{tcolorbox}%
}{}

\NewDocumentEnvironment{annotatedtext}{m m}{%
    \begin{tcolorbox}[
        annotated,
        colback = white, left = 0pt,
        before upper = {\panelheading{white}{white}{.}}
    ]
        #1
        \tcblower
        \small #2
    \end{tcolorbox}%
}{}

\subsection{Proof by Construction}
\label{subsections:techniques:existential:construction}



\paragraph{Description.}
A proof by construction proves a mathematical statement by finding %
an example that satisfies it. %
Precisely, it proves an existential statement, %
``$\exists s \in S$ such that $P(s)$,'' by defining a particular element %
$s_0 \in S$ and showing that $P(s_0)$ is true. %
Generally, a proof by construction mirrors the following format:
\begin{enumerate}{
    \item \textbf{Setup.} %
        Define a particular element $s_0$ such that $s_0 \in S$.
    \item \textbf{Verification.} %
        Show that $P(s_0)$ is true.
    \item \textbf{Conclusion.} %
        Conclude the proof, optionally by summarizing what has been found.
}
\end{enumerate}



\paragraph{Advice.}
\begin{itemize}{
    \item Use this proof only for an existential statement, %
        ``$\exists s \in S$ such that $P(s)$,'' %
        when an example can be found explicitly. %
        This proof cannot be used to prove a %
        \hyperref[sections:techniques:universal]{universal statement}, %
        since one example does not cover every element of $S$.
    \item Show both $s_0 \in S$ and $P(s_0)$, %
        as an element outside of $S$ does not prove the statement, %
        even if $P(s_0)$ is true.
}
\end{itemize}



\paragraph{Example.}
\begin{exercise}{
    Prove that $\exists z \in \Z$ such that $z^2 = 4$.
}
\end{exercise}
\begin{proof}{
    This statement can be proven by construction.

    \begin{annotatedpanel}[Setup.]{
        Let $z \in \Z$ such that $z = 2$.
    }{
        \textbf{Step 1.} Define $s_0$ such that $s_0 \in S$. %
        Here, $S = \Z$ and $s_0 = z = 2$, which is an integer.
    }
    \end{annotatedpanel}

    \begin{annotatedpanel}[Verification.]{
        Then $z^2 = 2^2 = 4$.
    }{
        \textbf{Step 2.} Show that $P(s_0)$ is true. %
        Here, $P(z)$ is the equation $z^2 = 4$.
    }
    \end{annotatedpanel}

    \begin{annotatedpanel}[Conclusion.]{
        Thus, $\exists z \in \Z$ such that $z^2 = 4$. %
        Therefore, the statement holds by construction. \qedhere
    }{
        \textbf{Step 3.} Conclude the proof. %
        Here, the summary states what has been found: %
        $z = 2$ is an integer such that $z^2 = 4$.
    }
    \end{annotatedpanel}
}
\end{proof}

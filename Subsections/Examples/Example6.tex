\subsection{Intersections of Open Sets}
\label{subsections:examples:example6}



\begin{exercise}{
    A set $S \subseteq \R$ is open if $\forall s \in S$, %
    $\exists \epsilon \in \R$ with $\epsilon > 0$ such that %
    $(s - \epsilon, s + \epsilon) \subseteq S$. %
    Prove that the intersection of two open sets is open.
}
\end{exercise}



\begin{proof}{
    This statement can be proven %
    \hyperref[subsections:techniques:universal:direct]{directly}.

    \begin{panel}[Setup.]{
        Let $S_1, S_2 \subseteq \R$ be open sets. %
        Let $s \in S_1 \cap S_2$. %
        Then $s \in S_1$ and $s \in S_2$. %
        We want to prove that %
        $\exists \epsilon \in \R$ with $\epsilon > 0$ such that %
        \[
            (s - \epsilon, s + \epsilon) \subseteq S_1 \cap S_2.
        \]
    }
    \end{panel}

    \begin{panel}[Applying the definitions.]{
        Since $S_1$ is open and $s \in S_1$, %
        $\exists \epsilon_1 \in \R$ with $\epsilon_1 > 0$ such that %
        \[
            (s - \epsilon_1, s + \epsilon_1) \subseteq S_1.
        \]
        Similarly, since $S_2$ is open and $s \in S_2$, %
        $\exists \epsilon_2 \in \R$ with $\epsilon_2 > 0$ such that %
        \[
            (s - \epsilon_2, s + \epsilon_2) \subseteq S_2.
        \]
    }
    \end{panel}

    \begin{panel}[Defining the minimum.]{
        Let $\epsilon = \min(\epsilon_1, \epsilon_2)$. %
        Then $\epsilon \in \R$ and $\epsilon > 0$ because %
        $\epsilon_1, \epsilon_2 \in \R$ and $\epsilon_1, \epsilon_2 > 0$. %
        Since $\epsilon \leq \epsilon_1$, %
        \[
            (s - \epsilon, s + \epsilon) \subseteq
            (s - \epsilon_1, s + \epsilon_1) \subseteq S_1.
        \]
        Similarly, since $\epsilon \leq \epsilon_2$, %
        \[
            (s - \epsilon, s + \epsilon) \subseteq
            (s - \epsilon_2, s + \epsilon_2) \subseteq S_2.
        \]
        Thus, %
        \[
            (s - \epsilon, s + \epsilon) \subseteq S_1 \cap S_2.
        \]
    }
    \end{panel}

    Thus, $S_1 \cap S_2$ is open. %
    Therefore, the statement holds.
}
\end{proof}

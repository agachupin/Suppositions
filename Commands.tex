% Colors
% Link gold is old gold darkened for legible text on white.
\definecolor{oldgold}{HTML}{9E7E38}
\colorlet{linkgold}{oldgold!85!black}



% Text
\newcommand{\titleuppercase}[1]{\textls[50]{\MakeUppercase{#1}}}



% Headings
% Each numbered section opens on its own page, centered vertically.
\titleformat{\section}[block]%
    {\large\bfseries\filcenter}%
    {\mdseries\thesection}%
    {1em}%
    {\titleuppercase}

\let\sectionstyle\section
\RenewDocumentCommand{\section}{s o m}{%
    \IfBooleanTF{#1}{%
        \sectionstyle*{#3}%
    }{%
        \clearpage
        \thispagestyle{empty}%
        \vspace*{\fill}%
        \IfNoValueTF{#2}{\sectionstyle{#3}}{\sectionstyle[#2]{#3}}%
        \vspace*{\fill}%
    }%
}

\titleformat{\subsection}[block]%
    {\normalsize\bfseries}%
    {\mdseries\thesubsection}%
    {1em}%
    {\titleuppercase}

% Panels
% A panel is a shaded box with a dark left border and an optional heading.
% An annotated panel adds a gold column on the right for the annotation.
\newlength{\annotationwidth}
\setlength{\annotationwidth}{0.36\textwidth}
\newlength{\annotationgap}
\setlength{\annotationgap}{32pt}

% Heading text, rule, and spacing.
% #1: heading color, #2: rule color, #3: heading text.
\newcommand{\panelheading}[3]{%
    {\small\bfseries\color{#1}\titleuppercase{#3}\par}%
    \nointerlineskip
    \vspace{7pt}%
    {\color{#2}\hrule height 0.4pt}%
    \nointerlineskip
    \vspace{7pt}%
}

\tcbset{
    panelbase/.style = {
        enhanced, unbreakable,
        frame hidden, boxrule = 0pt, sharp corners,
        colback = black!3,
        borderline west = {1pt}{0pt}{black!75},
        left = 16pt, right = 16pt, top = 12pt, bottom = 12pt,
        before skip = 14pt, after skip = 14pt
    },
    annotated/.style = {
        panelbase,
        sidebyside, sidebyside align = top, lower separated = false,
        sidebyside gap = \annotationgap,
        righthand width = \annotationwidth,
        underlay = {%
            \begin{tcbclipinterior}
                \fill[oldgold!15]
                    ([xshift = -\annotationwidth - \annotationgap]%
                        interior.north east)
                    rectangle (interior.south east);
            \end{tcbclipinterior}%
        },
        before lower = {%
            \panelheading{oldgold}{oldgold!40}{Annotation}%
        }
    }
}

% Usage: \begin{panel}[Heading]{Content}\end{panel}
\NewTColorBox{panel}{o}{
    panelbase,
    before upper = {%
        \IfValueT{#1}{\panelheading{black!70}{black!20}{#1}}%
    }
}

% Usage: \begin{annotatedpanel}[Heading]{Content}{Annotation}
%        \end{annotatedpanel}
\NewDocumentEnvironment{annotatedpanel}{o m m}{%
    \begin{tcolorbox}[
        annotated,
        before upper = {%
            \IfValueT{#1}{\panelheading{black!70}{black!20}{#1}}%
        }
    ]
        #2
        \tcblower
        \small #3
    \end{tcolorbox}%
}{}

\subsection{Dot Products of Factorials}
\label{subsections:examples:example4}



\begin{exercise}{
    For %
    $(a_1, a_2, a_3, \dotsc, a_n), (b_1, b_2, b_3, \dotsc, b_n) \in \R^n$,
    the dot product is %
    \[
        (a_1, a_2, a_3, \dotsc, a_n) \cdot (b_1, b_2, b_3, \dotsc, b_n) =
        a_1 b_1 + a_2 b_2 + a_3 b_3 + \dotsb + a_n b_n.
    \]
    For example, %
    \[
        (5, 2, -3) \cdot (3, 7, 4) = 5 \cdot 3 + 2 \cdot 7 + (-3) \cdot 4 = 17.
    \]
    Prove that $\forall n \in \Z$ with $n > 0$, %
    \[
        (1, 2, 3, \dotsc, n) \cdot (1!, 2!, 3!, \dotsc, n!) = (n + 1)! - 1.
    \]
}
\end{exercise}



\begin{proof}{
    This statement can be proven by %
    \hyperref[subsections:techniques:universal:induction]
        {mathematical induction}.

    \begin{panel}[Setup.]{
        Let $P(n)$ be the statement that %
        \[
            (1, 2, 3, \dotsc, n) \cdot (1!, 2!, 3!, \dotsc, n!) = (n + 1)! - 1.
        \]
        Since $n \in \Z$ and $n > 0$, the smallest case is $n_0 = 1$.
    }
    \end{panel}

    \begin{panel}[Base Case: Let $n = 1$.]{
        Then %
        \begin{align*}
            (1, 2, 3, \dotsc, n) \cdot (1!, 2!, 3!, \dotsc, n!)
                &= (1) \cdot (1!)\\
                &= 1 \cdot 1\\
                &= 1,
        \end{align*}
        and %
        \begin{align*}
            (n + 1)! - 1
                &= (1 + 1)! - 1\\
                &= 2! - 1\\
                &= 2 - 1\\
                &= 1.
        \end{align*}
        Thus, both sides equal $1$, so $P(1)$ is true.
    }
    \end{panel}

    \begin{panel}[Inductive Hypothesis.]{
        Let $n \in \Z$ such that $n \ge 1$ and $P(n)$ is true, that is, %
        \[
            (1, 2, 3, \dotsc, n) \cdot (1!, 2!, 3!, \dotsc, n!) = (n + 1)! - 1.
        \]
    }
    \end{panel}

    \begin{panel}[Inductive Step: Consider $P(n + 1)$.]{
        We want to prove that %
        \[
            (1, 2, 3, \dotsc, n, n + 1) \cdot
            (1!, 2!, 3!, \dotsc, n!, (n + 1)!) = [(n + 1) + 1]! - 1.
        \]
        By the definition of the dot product, %
        \begin{align*}
            (1, 2, 3, \dotsc, n, n + 1) \cdot
            (1!, 2!, 3!, \dotsc, n!, (n + 1)!)
                &= (1, 2, 3, \dotsc, n) \cdot (1!, 2!, 3!, \dotsc, n!)\\
                &\quad + (n + 1)(n + 1)!.
        \end{align*}
        By the inductive hypothesis, %
        \begin{align*}
            (1, 2, 3, \dotsc, n, n + 1) \cdot
            (1!, 2!, 3!, \dotsc, n!, (n + 1)!)
                &= [(n + 1)! - 1] + (n + 1)(n + 1)!\\
                &= (n + 1)! + (n + 1)(n + 1)! - 1\\
                &= (n + 1)! \cdot [1 + (n + 1)] - 1\\
                &= (n + 1)! \cdot (n + 2) - 1\\
                &= (n + 2)! - 1\\
                &= [(n + 1) + 1]! - 1.
        \end{align*}
        Thus, $P(n + 1)$ is true.
    }
    \end{panel}

    Thus, the statement holds for $n = 1$, %
    and if it holds for $n$, then it holds for $n + 1$. %
    Therefore, the statement holds by the principle of mathematical induction.
}
\end{proof}

\subsection{Distances from a Fixed Point}
\label{subsections:examples:example2}



\begin{exercise}{
    Prove that if $x, y \in \R$ with $x < -4$ and $y > 2$, %
    then the distance from $(x, y)$ to $(1, -2)$ is at least $6$.
}
\end{exercise}



\begin{proof}{
    This statement can be proven %
    \hyperref[subsections:techniques:universal:direct]{directly}.

    \begin{panel}[Setup.]{
        Let $x, y \in \R$ with $x < -4$ and $y > 2$. %
        Let $d$ denote the distance from $(x, y)$ to $(1, -2)$. %
        By the Euclidean distance formula, %
        \[
            d = \sqrt{(1 - x)^2 + (-2 - y)^2}.
        \]
    }
    \end{panel}

    \begin{panel}[Bound 1: We know $x < -4$.]{
        Then %
        \begin{align*}
                   -x &> 4\\
               -x + 1 &> 4 + 1\\
                1 - x &> 5.
        \end{align*}
        Since both sides are positive, squaring preserves the inequality. %
        Thus, %
        \[
            (1 - x)^2 > 5^2 = 25.
        \]
    }
    \end{panel}

    \begin{panel}[Bound 2: We know $y > 2$.]{
        Then %
        \begin{align*}
                    -y &< -2\\
                -y - 2 &< -2 - 2\\
                -2 - y &< -4.
        \end{align*}
        Since both sides are negative, squaring reverses the inequality. %
        Thus, %
        \[
            (-2 - y)^2 > (-4)^2 = 16.
        \]
    }
    \end{panel}

    \begin{panel}[Combining the bounds.]{
        By Bounds 1 and 2, %
        \begin{align*}
            d &= \sqrt{(1 - x)^2 + (-2 - y)^2}\\
              &> \sqrt{25 + 16}\\
              &= \sqrt{41}\\
              &> \sqrt{36}\\
              &= 6.
        \end{align*}
    }
    \end{panel}

    Thus, the distance from $(x, y)$ to $(1, -2)$ is at least $6$. %
    Therefore, the statement holds.
}
\end{proof}

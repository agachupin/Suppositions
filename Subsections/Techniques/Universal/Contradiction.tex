\subsection{Proof by Contradiction}
\label{subsections:techniques:universal:contradiction}



\paragraph{Description.}
A proof by contradiction proves a mathematical statement by showing that %
its negation cannot be true. %
Precisely, it proves a statement $P$ by assuming $\neg P$ and deriving a %
contradiction, meaning a statement $C$ such that $C$ and $\neg C$ are %
both true. %
This works because a statement is either true or false: %
if $\neg P$ is false, then $P$ is true. %
Generally, a proof by contradiction mirrors the following format:
\begin{enumerate}{
    \item \textbf{Setup.} %
        Assume that $\neg P$ is true.
    \item \textbf{Verification.} %
        Derive a contradiction by showing that $C$ and $\neg C$ are both %
        true.
    \item \textbf{Conclusion.} %
        Conclude that $\neg P$ is false and therefore that $P$ is true, %
        optionally by summarizing what has been found.
}
\end{enumerate}



\paragraph{Advice.}
\begin{itemize}{
    \item Use this proof when assuming $\neg P$ gives something concrete %
        to work with, such as when $P$ claims that something does not exist.
    \item State the assumption $\neg P$ explicitly at the start of the %
        proof, for example with ``Suppose, for contradiction, that $\dots$,'' %
        as the purpose of the proof can otherwise be unclear.
}
\end{itemize}



\paragraph{Example.}
\begin{exercise}{
    Prove that $\neg\exists z \in \Z$ such that $2z = 1$.
}
\end{exercise}
\begin{proof}{
    This statement can be proven by contradiction.

    \begin{annotatedpanel}[Setup.]{
        Suppose, for contradiction, that $\exists z \in \Z$ such that %
        $2z = 1$.
    }{
        \textbf{Step 1.} Assume that $\neg P$ is true. %
        Here, $P$ is the statement $\neg\exists z \in \Z$ such that %
        $2z = 1$, so $\neg P$ is the statement $\exists z \in \Z$ such that %
        $2z = 1$.
    }
    \end{annotatedpanel}

    \begin{annotatedpanel}[Verification.]{
        Then $z = \frac{1}{2}$, so $z \notin \Z$. %
        This contradicts $z \in \Z$.
    }{
        \textbf{Step 2.} Derive a contradiction by showing that $C$ and %
        $\neg C$ are both true. %
        Here, $C$ is the statement $z \in \Z$ and $\neg C$ is the %
        statement $z \notin \Z$, which follows from dividing $2z = 1$ %
        by $2$.
    }
    \end{annotatedpanel}

    \begin{annotatedpanel}[Conclusion.]{
        Thus, the assumption that $\exists z \in \Z$ such that $2z = 1$ %
        is false, so $\neg\exists z \in \Z$ such that $2z = 1$. %
        Therefore, the statement holds by contradiction. \qedhere
    }{
        \textbf{Step 3.} Conclude that $\neg P$ is false and therefore that %
        $P$ is true. %
        Here, the summary states what has been found: %
        no integer $z$ satisfies $2z = 1$.
    }
    \end{annotatedpanel}
}
\end{proof}

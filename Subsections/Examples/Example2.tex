\subsection{Distances from a Fixed Point}
\label{subsections:examples:example2}



\begin{exercise}{
    Prove that if $x < -4$ and $y > 2$, then the distance from $(x, y)$ to %
    $(1, -2)$ is at least $6$.
}
\end{exercise}



\begin{proof}{
    This statement can be proven %
    \hyperref[subsections:techniques:universal:direct]{directly}.

    Let $x, y \in \R$ such that $x < -4$ and $y > 2$. Let $d$ denote the %
    distance from $(x, y)$ to $(1, -2)$. By the Euclidean distance formula, %
    $d = \sqrt{(1 - x)^2 + (-2 - y)^2}$.

    \begin{panel}[Bound 1: We know $x < -4$.]{
        Then
        \begin{align*}{
            -x &> 4 \\
            -x + 1 &> 4 + 1 \\
            1 - x &> 5 \\
            (1 - x)^2 &> 5^2 \\
            (1 - x)^2 &> 25.
        }
        \end{align*}
    }
    \end{panel}
    \begin{panel}[Boundd 2: We know $y > 2$.]{
        Then
        \begin{align*}{
            -y &< -2 \\
            -y - 2 &< -2 - 2 \\
            -2 - y &< -4 \\
            (-2 - y)^2 &> (-4)^2 \\
            (-2 - y)^2 &> 16.
        }
        \end{align*}
    }
    \end{panel}
    \begin{panel}[Combining the bounds.]{
        Then
        \begin{align*}{
            d &= \sqrt{(1 - x)^2 + (-2 - y)^2} \\
              &> \sqrt{25 + 16} \\
              &= \sqrt{41} \\
              &> 6.
        }
        \end{align*}
    }
    \end{panel}

    Thus, $d \geq 6$. Therefore, the statement holds.
}
\end{proof}
